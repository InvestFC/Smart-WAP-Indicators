//@version=6
// Smart WAP Crypto: trend and continuation signals with simulated trade levels for 24/7 crypto markets. Defaults are tuned for 15m charts. Levels and exits are informational only, no real position exists.
// Developed by @bearhugs26 for Worth A Paisa on Telegram (for free!)
indicator("Smart WAP Crypto", "WAP Crypto", overlay=true, max_labels_count=500, max_lines_count=500)

// ATR
atrPeriod = input.int(14, "ATR Period", minval=1, group="ATR Trailing Stop")
atrMultiplier = input.float(4.0, "ATR Multiplier", step=0.1, minval=0, group="ATR Trailing Stop", tooltip="Trailing-stop distance in ATRs. Higher = fewer, slower trend flips.")

atr = ta.atr(atrPeriod)
atrStop = atrMultiplier * atr

var float atrTrail = na
if na(atrTrail[1])
    atrTrail := close - atrStop
else if close > atrTrail[1] and close[1] > atrTrail[1]
    atrTrail := math.max(atrTrail[1], close - atrStop)
else if close < atrTrail[1] and close[1] < atrTrail[1]
    atrTrail := math.min(atrTrail[1], close + atrStop)
else if close > atrTrail[1]
    atrTrail := close - atrStop
else
    atrTrail := close + atrStop

var int atrRegime = 0
if na(atrRegime[1])
    atrRegime := 0
else if close[1] < atrTrail[1] and close > atrTrail[1]
    atrRegime := 1
else if close[1] > atrTrail[1] and close < atrTrail[1]
    atrRegime := -1
else
    atrRegime := atrRegime[1]

bullFlip = barstate.isconfirmed and atrRegime == 1 and atrRegime[1] == -1
bearFlip = barstate.isconfirmed and atrRegime == -1 and atrRegime[1] == 1

var int tradeDir = 0
var float tradeEntry = na
var float tradeSL = na
var float tradeBE = na
var int tradeEntryBar = na
var bool t1Hit = false
var float bookedFrac = 0.0
var float bookedR = 0.0
var int tradeCount = 0

// Booking model: touching a target books 25% of the position there, and the last target closes
// whatever remains. Once the first target is touched, the stop moves to the entry plus the
// Breakeven Buffer, and the rest closes at that stop, at an ATR flip or at the last target. The
// result of a closed trade in R blends the booked parts with the exit of the remainder.

// Result of a closed trade in R, where 1R is the initial stop distance (entry to stop).
rResult(float exitPrice, int dir) =>
    float risk = math.abs(tradeEntry - tradeSL)
    risk > 0 ? dir * (exitPrice - tradeEntry) / risk : na

bookedText(float frac) =>
    frac > 0 ? "\n" + str.tostring(math.round(frac * 100)) + "% booked" : ""

// Captured before any closure path (ATR flip, SL, T4) or target check runs this bar.
wasOpen = tradeDir != 0
t1WasHit = t1Hit

bool flipClosed = false
int flipCloseDir = 0
float flipR = na
float flipBooked = na
int flipTradeNum = na

// Before the first target is touched, a candle that closes beyond the stop closes the trade at that close,
// even if the same candle also closes across the ATR trail, so the flip does not close it here.
bool stopCrossed = tradeDir != 0 and not t1Hit and bar_index > tradeEntryBar and (tradeDir == 1 ? close <= tradeSL : close >= tradeSL)

if bullFlip and tradeDir == -1 and not stopCrossed
    flipR := bookedR + (1 - bookedFrac) * rResult(close, tradeDir)
    flipBooked := bookedFrac
    flipTradeNum := tradeCount
    tradeDir := 0
    flipClosed := true
    flipCloseDir := 1

if bearFlip and tradeDir == 1 and not stopCrossed
    flipR := bookedR + (1 - bookedFrac) * rResult(close, tradeDir)
    flipBooked := bookedFrac
    flipTradeNum := tradeCount
    tradeDir := 0
    flipClosed := true
    flipCloseDir := -1

// Each regime has its own line, and a line breaks where its regime ends. A flip therefore shows
// a clean step to the other side in the other colour, with no diagonal connecting the two.
plot(atrRegime == 1 ? atrTrail : na, "ATR Trailing Stop (up)", color=color.green, linewidth=1, style=plot.style_linebr)
plot(atrRegime == -1 ? atrTrail : na, "ATR Trailing Stop (down)", color=color.red, linewidth=1, style=plot.style_linebr)
plot(atrRegime == 0 ? atrTrail : na, "ATR Trailing Stop (before first flip)", color=color.navy, linewidth=1, style=plot.style_linebr)

// MOVING AVERAGES
maType = input.string("EMA", "MA Calculation", options=["EMA", "SMA", "WMA", "RMA", "VWMA", "HMA"], group="Moving Averages", tooltip="Applies to the Fast/Slow pair (which drives MA Regime Confirmation and Continuation's pullback/resume logic) AND the additional 20/50/100/200 MAs. Changing it away from EMA changes the signal logic, not just the chart overlay. VWMA needs volume data and returns nothing on symbols without it.")
fastLen = input.int(5, "Fast MA Length", minval=1, group="Moving Averages")
slowLen = input.int(15, "Slow MA Length", minval=1, group="Moving Averages")
showExtraMAs = input.bool(false, "Show Additional MAs (20/50/100/200)", group="Moving Averages", tooltip="Reference MAs only, not part of the signal logic.")
showFastSlow = input.bool(false, "Show Fast/Slow MAs", group="Moving Averages", tooltip="Shows the Fast and Slow MA lines. They drive MA Regime Confirmation and the Continuation signal whether shown or not.")
fastColor = input.color(color.yellow, "Fast MA", inline="maColours1", group="Moving Averages")
slowColor = input.color(color.purple, "Slow MA", inline="maColours1", group="Moving Averages")
ma20Color = input.color(color.aqua, "20 MA", inline="maColours2", group="Moving Averages")
ma50Color = input.color(color.blue, "50 MA", inline="maColours2", group="Moving Averages")
ma100Color = input.color(color.orange, "100 MA", inline="maColours2", group="Moving Averages")
ma200Color = input.color(color.red, "200 MA", inline="maColours2", group="Moving Averages")

ma(series float source, int length) =>
    float result = na
    if maType == "EMA"
        result := ta.ema(source, length)
    else if maType == "SMA"
        result := ta.sma(source, length)
    else if maType == "WMA"
        result := ta.wma(source, length)
    else if maType == "RMA"
        result := ta.rma(source, length)
    else if maType == "VWMA"
        result := ta.vwma(source, length)
    else
        result := ta.hma(source, length)
    result

fastMA = ma(close, fastLen)
slowMA = ma(close, slowLen)
ma20 = showExtraMAs ? ma(close, 20) : na
ma50 = showExtraMAs ? ma(close, 50) : na
ma100 = showExtraMAs ? ma(close, 100) : na
ma200 = showExtraMAs ? ma(close, 200) : na

plot(showFastSlow ? fastMA : na, "Fast MA", color=fastColor, linewidth=2)
plot(showFastSlow ? slowMA : na, "Slow MA", color=slowColor, linewidth=2)
plot(ma20, "20 MA", color=ma20Color)
plot(ma50, "50 MA", color=ma50Color)
plot(ma100, "100 MA", color=ma100Color)
plot(ma200, "200 MA", color=ma200Color, linewidth=2)

// SIGNAL INPUTS
smartBuyOn = input.bool(true, "Smart Buy Signal", group="Signals")
smartSellOn = input.bool(true, "Smart Sell Signal", group="Signals")
showFlipLabels = input.bool(true, "Show ATR Flip Labels", group="Signals", tooltip="Label shown when an ATR flip closes an open simulated trade. It shows whether the trade ended in a profit or a loss.")
maxLosses = input.int(0, "Max Losing Trades Per Day", minval=0, group="Signals", tooltip="0 turns this off. When set, new signals and their entry alerts stop for the rest of the day (exchange time) once this many simulated trades have closed at a net loss, whether by the stop or an ATR flip. The label that shows how the limit-reaching trade ended adds Done for the day. For example, 2 stops new entries after two losing trades.")

// MA CONFIRMATION
useMAConfirm = input.bool(true, "Use MA Fast/Slow Regime Confirmation", group="MA Regime Confirmation", tooltip="After an ATR flip, wait for the Fast MA to be on the right side of the Slow MA before signalling (a state check, not a crossover). The flip bar itself cannot confirm; the window starts on the bar after it. Off = signal on the flip bar itself.")
maxPendingBars = input.int(10, "Maximum Pending Confirmation Bars", minval=1, group="MA Regime Confirmation", tooltip="How many bars to wait for MA confirmation after an ATR flip before giving up.")

// RSI / COOLDOWN
useRSI = input.bool(true, "Use RSI Filter", group="Signal Filter", tooltip="Buy needs RSI at or above the Buy level. Sell needs RSI at or below the Sell level.")
rsiLen = input.int(14, "RSI Length", minval=1, group="Signal Filter")
buyRSI = input.int(55, "Buy RSI Level", minval=0, maxval=100, group="Signal Filter")
sellRSI = input.int(45, "Sell RSI Level", minval=0, maxval=100, group="Signal Filter")
minSignalGap = input.int(10, "Minimum Bars Between Signals", minval=0, group="Signal Filter", tooltip="Minimum bars between any two signals (trend or continuation).")

rsi = ta.rsi(close, rsiLen)

// STOP AND TARGETS (the stop is ATR-based and every target is a multiple of the stop distance; informational only, not orders)
slMult = input.float(3.5, "Stop Loss ATR Multiplier", step=0.1, minval=0, group="SL / Targets", tooltip="Stop distance in ATRs from the signal bar's close. Each target sits at its Risk Reward multiple of this distance.")
beBuffer = input.float(0.2, "Breakeven Buffer (x ATR)", step=0.05, minval=0, group="SL / Targets", tooltip="When the first target is touched, the stop moves to the entry plus this many ATR (minus for a SELL), using the ATR frozen at entry, and stays there until the last target. Like the initial stop, it closes the trade only when a candle closes beyond it.")
t1RR = input.float(1.0, "Target 1 Risk Reward", step=0.1, minval=0, group="SL / Targets")
t2RR = input.float(2.5, "Target 2 Risk Reward", step=0.1, minval=0, group="SL / Targets")
t3RR = input.float(4.0, "Target 3 Risk Reward", step=0.1, minval=0, group="SL / Targets")
t4RR = input.float(5.5, "Target 4 Risk Reward", step=0.1, minval=0, group="SL / Targets", tooltip="Target 4 is the last target. Touching it closes the simulated trade.")
showTargets = input.bool(true, "Show Target/SL Labels and Lines", group="SL / Targets", tooltip="Levels are informational only. The entry line marks the confirmed signal bar's close. The stop sits the Stop Loss ATR Multiplier times ATR from that close, and each target sits at its Risk Reward multiple of that distance. Continuation buys and sells use the Continuation Entry Levels settings instead. Levels are frozen for that simulated trade. Touching a target books 25% of the position there and ticks its label, and the last target closes the rest. Before the first target, the trade closes when a candle closes beyond the stop; a wick through it does not count. After the first target, the stop moves to the entry plus the Breakeven Buffer, and an ATR flip also closes what remains. T1 and T2 are drawn at entry, and T3 appears when T1 is touched and T4 when T2 is touched. A closed trade keeps its lines and only the label that shows how it ended.")
eraseOld = input.bool(true, "Erase Older Trade Drawings", group="SL / Targets", tooltip="Deletes the level lines, price labels and exit label of older simulated trades. BUY, SELL, C BUY, C SELL and ADD labels always stay.")
keepTrades = input.int(4, "Trades to Keep Drawn", minval=4, group="SL / Targets", tooltip="How many of the most recent trades keep their drawings, counting an open trade. 4 keeps a trend entry and its 3 continuation entries when they all belong to one trend. Raise it if you raise Max Continuation Re-Entries.")

// CONTINUATION ENTRY LEVELS (stop and targets set directly in ATRs from the entry; informational only, not orders)
contSlMult = input.float(4.5, "Stop Loss ATR Multiplier", step=0.1, minval=0, group="Continuation Entry Levels", tooltip="Used only for continuation buys and sells. Stop distance in ATRs from the signal bar's close.")
contT1Mult = input.float(1.5, "Target 1 ATR Multiplier", step=0.1, minval=0, group="Continuation Entry Levels", tooltip="Used only for continuation buys and sells. Each target is a distance in ATRs from the signal bar's close.")
contT2Mult = input.float(2.7, "Target 2 ATR Multiplier", step=0.1, minval=0, group="Continuation Entry Levels")
contT3Mult = input.float(4.7, "Target 3 ATR Multiplier", step=0.1, minval=0, group="Continuation Entry Levels")
contT4Mult = input.float(6.5, "Target 4 ATR Multiplier", step=0.1, minval=0, group="Continuation Entry Levels", tooltip="Target 4 is the last target. Touching it closes the simulated trade.")

// Text for a level's price label: the name and price while the level is live, or the name and a mark once a
// target is touched (tick) or a stop closes the trade (cross).
levelText(string name, string mark, float price) =>
    mark == "" ? name + ": " + str.tostring(price, format.mintick) : name + mark

// Price label at the right end of a level line. It moves along with the line as the line extends.
levelLabel(float y, string txt, color c) =>
    label.new(bar_index, y, txt, style=label.style_label_left, color=color.new(c, 20), textcolor=color.white, size=size.tiny)

// STATE
var float tradeT1 = na
var float tradeT2 = na
var float tradeT3 = na
var float tradeT4 = na

var bool t2Hit = false
var bool t3Hit = false
var bool t4Hit = false

var line entryLine = na
var line slLine = na
var line beLine = na
var line t1Line = na
var line t2Line = na
var line t3Line = na
var line t4Line = na

var label entryLbl = na
var label slLbl = na
var label t1Lbl = na
var label t2Lbl = na
var label t3Lbl = na
var label t4Lbl = na

var int pendingDir = 0
var int pendingBar = na

var bool longPullbackSeen = false
var bool shortPullbackSeen = false
var int lastFlipDir = 0
var int continuationCount = 0
// ADD signals given to the open trade; reset at every entry.
var int addCount = 0

var int lastSignalBar = na

// CONTINUATION INPUTS
useContinuation = input.bool(true, "Enable Continuation Signal", group="Continuation Signal", tooltip="Re-enters a still-running trend after a pullback to the Fast MA followed by a resume candle, without needing a fresh ATR flip. Continuation entries are labelled C BUY and C SELL. When a simulated trade is already open in the same direction, the same setup shows ADD instead: it opens no new trade, leaves the open trade's levels and exits unchanged, and uses one of the re-entries. With Show Discretionary Aids on, the label adds the trend panel's reading on the signal candle: Strong (solid), Neutral for Building (lighter) or Weak (faded). With it off the label is plain and solid. C BUY, C SELL and ADD labels also show HTF ✓ when the higher timeframe's ATR trend agrees with the signal.")
maxReentries = input.int(3, "Max Continuation Re-Entries Per Regime", minval=1, group="Continuation Signal", tooltip="Cap on continuation entries, ADD signals included, within one ATR regime. Resets on every ATR flip.")
maxAdds = input.int(1, "Max ADDs Per Trade", minval=0, group="Continuation Signal", tooltip="How many ADD signals one open simulated trade can get. Set this to your own risk appetite: every add you take raises your real risk on the trade above what its stop and targets show. 0 turns ADD off. ADD signals also count toward Max Continuation Re-Entries Per Regime.")

// FLIPS
if bullFlip
    lastFlipDir := 1
    continuationCount := 0
    longPullbackSeen := false
    shortPullbackSeen := false
    if useMAConfirm
        pendingDir := 1
        pendingBar := bar_index
else if bearFlip
    lastFlipDir := -1
    continuationCount := 0
    longPullbackSeen := false
    shortPullbackSeen := false
    if useMAConfirm
        pendingDir := -1
        pendingBar := bar_index

// MA PENDING WINDOW
pendingAge = pendingDir != 0 and not na(pendingBar) ? bar_index - pendingBar : na

if pendingDir != 0 and pendingAge > maxPendingBars
    pendingDir := 0
    pendingBar := na

bool longMAReady = bullFlip
bool shortMAReady = bearFlip

if useMAConfirm
    longMAReady := pendingDir == 1 and pendingAge >= 1 and fastMA > slowMA
    shortMAReady := pendingDir == -1 and pendingAge >= 1 and fastMA < slowMA

longRSI = not useRSI or rsi >= buyRSI
shortRSI = not useRSI or rsi <= sellRSI

// CONTINUATION: pullback to the Fast MA, then a resume candle
longPullback = barstate.isconfirmed and atrRegime == 1 and lastFlipDir == 1 and pendingDir != 1 and low <= fastMA and close < fastMA
shortPullback = barstate.isconfirmed and atrRegime == -1 and lastFlipDir == -1 and pendingDir != -1 and high >= fastMA and close > fastMA

longResume = barstate.isconfirmed and atrRegime == 1 and lastFlipDir == 1 and pendingDir != 1 and close > fastMA and close > open
shortResume = barstate.isconfirmed and atrRegime == -1 and lastFlipDir == -1 and pendingDir != -1 and close < fastMA and close < open

if longPullback
    longPullbackSeen := true

if shortPullback
    shortPullbackSeen := true

// [1] reads the previous bar's value, so a flip must also block it explicitly.
prevLongPullback = longPullbackSeen[1] and not bullFlip
prevShortPullback = shortPullbackSeen[1] and not bearFlip

// LOSING TRADES PER DAY
// Simulated trades closed at a net loss in the current day. An ATR flip exit is counted here, before the
// signals, so it also blocks a re-entry on the same candle. Stop exits are counted where they happen.
var int lossCount = 0
if timeframe.change("D")
    lossCount := 0
if flipClosed and flipR < 0
    lossCount += 1
// True when this candle's ATR flip exit is the losing trade that reaches the limit.
bool flipReachedLimit = maxLosses > 0 and flipClosed and flipR < 0 and lossCount == maxLosses
bool lossLimitOK = maxLosses == 0 or lossCount < maxLosses

// SIGNALS
tradeFlat = tradeDir == 0
cooldownOK = na(lastSignalBar) or bar_index - lastSignalBar >= minSignalGap

trendBuy = barstate.isconfirmed and lossLimitOK and smartBuyOn and longMAReady and longRSI and tradeFlat and cooldownOK and not na(atr)
trendSell = barstate.isconfirmed and lossLimitOK and smartSellOn and shortMAReady and shortRSI and tradeFlat and cooldownOK and not na(atr)

contBuy = barstate.isconfirmed and lossLimitOK and useContinuation and smartBuyOn and tradeFlat and not na(atr) and prevLongPullback and longResume and continuationCount < maxReentries and longRSI and cooldownOK
contSell = barstate.isconfirmed and lossLimitOK and useContinuation and smartSellOn and tradeFlat and not na(atr) and prevShortPullback and shortResume and continuationCount < maxReentries and shortRSI and cooldownOK

// Trend and continuation cannot both be true (continuation requires no pending confirmation),
// but if opposite signals ever coincide, neither fires.
buyCand = trendBuy or contBuy
sellCand = trendSell or contSell
buySignal = buyCand and not sellCand
sellSignal = sellCand and not buyCand

trendBuySig = buySignal and trendBuy
trendSellSig = sellSignal and trendSell
contBuySig = buySignal and contBuy and not trendBuy
contSellSig = sellSignal and contSell and not trendSell

if buySignal or sellSignal
    lastSignalBar := bar_index

if trendBuySig or trendSellSig
    pendingDir := 0
    pendingBar := na

if contBuySig
    continuationCount += 1
    longPullbackSeen := false

if contSellSig
    continuationCount += 1
    shortPullbackSeen := false

// TRADE DRAWING HISTORY
// Each trade's drawings are stored with the trade's number so the oldest trades can be deleted.
var array<line> drawnLines = array.new<line>()
var array<int> lineOwner = array.new<int>()
var array<label> drawnLabels = array.new<label>()
var array<int> labelOwner = array.new<int>()

keepLine(line l) =>
    if eraseOld
        array.push(drawnLines, l)
        array.push(lineOwner, tradeCount)

keepLabel(label l, int owner) =>
    if eraseOld
        array.push(drawnLabels, l)
        array.push(labelOwner, owner)

// Deletes the drawings of every trade older than the most recent Trades to Keep Drawn.
pruneDrawings() =>
    int oldest = tradeCount - keepTrades + 1
    if array.size(drawnLines) > 0
        for i = array.size(drawnLines) - 1 to 0
            if array.get(lineOwner, i) < oldest
                line.delete(array.remove(drawnLines, i))
                array.remove(lineOwner, i)
    if array.size(drawnLabels) > 0
        for i = array.size(drawnLabels) - 1 to 0
            if array.get(labelOwner, i) < oldest
                label.delete(array.remove(drawnLabels, i))
                array.remove(labelOwner, i)

// EXTEND TRADE LEVELS
// Moves the trade's lines and labels to the current candle. The stop line ends on the candle where
// the first target is touched, and the dotted breakeven line starts there.
extendLevels(bool withStop) =>
    if withStop
        line.set_x2(slLine, bar_index)
        label.set_x(slLbl, bar_index)
    if not na(beLine)
        line.set_x2(beLine, bar_index)
        label.set_x(slLbl, bar_index)
    line.set_x2(entryLine, bar_index)
    line.set_x2(t1Line, bar_index)
    line.set_x2(t2Line, bar_index)
    if not na(t3Line)
        line.set_x2(t3Line, bar_index)
    if not na(t4Line)
        line.set_x2(t4Line, bar_index)
    label.set_x(entryLbl, bar_index)
    label.set_x(t1Lbl, bar_index)
    label.set_x(t2Lbl, bar_index)
    if not na(t3Lbl)
        label.set_x(t3Lbl, bar_index)
    if not na(t4Lbl)
        label.set_x(t4Lbl, bar_index)

// Deletes a closed trade's price labels except the one that shows how it ended: the stop label for a stop
// exit, the last target's label for a last-target exit, none for an ATR flip exit, which has its own exit
// label. The lines stay.
clearLevelLabels(bool keepStop, bool keepLast) =>
    label.delete(entryLbl)
    if not keepStop
        label.delete(slLbl)
    label.delete(t1Lbl)
    label.delete(t2Lbl)
    label.delete(t3Lbl)
    if not keepLast
        label.delete(t4Lbl)

// Lines are moved only on candles where their end point matters: the closing candle, the candle
// that touches the first target, and the latest candle. An ATR flip close is handled here, before
// the entry block, because a same-candle re-entry replaces the line variables with the new trade's.
if showTargets and wasOpen and flipClosed and tradeEntryBar < bar_index and not na(slLine)
    extendLevels(not t1WasHit)
    clearLevelLabels(false, false)

// ENTRY (entry reference = the confirmed signal bar's close; ATR frozen at the same moment)
if buySignal or sellSignal
    int entryDir = buySignal ? 1 : -1

    tradeDir := entryDir
    tradeEntry := close
    tradeEntryBar := bar_index
    tradeCount += 1
    addCount := 0

    t1Hit := false
    t2Hit := false
    t3Hit := false
    t4Hit := false
    bookedFrac := 0.0
    bookedR := 0.0

    // Continuation entries set the stop and targets directly in ATRs. Trend entries set targets as
    // multiples of the stop distance.
    bool contLevels = contBuySig or contSellSig
    float riskDist = atr * (contLevels ? contSlMult : slMult)
    tradeSL := tradeEntry - entryDir * riskDist
    tradeBE := tradeEntry + entryDir * atr * beBuffer
    beLine := na
    tradeT1 := tradeEntry + entryDir * (contLevels ? atr * contT1Mult : riskDist * t1RR)
    tradeT2 := tradeEntry + entryDir * (contLevels ? atr * contT2Mult : riskDist * t2RR)
    tradeT3 := tradeEntry + entryDir * (contLevels ? atr * contT3Mult : riskDist * t3RR)
    tradeT4 := tradeEntry + entryDir * (contLevels ? atr * contT4Mult : riskDist * t4RR)

    // Only the next two targets are drawn: T3 appears when T1 is touched and T4 when T2 is touched.
    t3Line := na
    t4Line := na
    t3Lbl := na
    t4Lbl := na

    if showTargets
        entryLine := line.new(bar_index, tradeEntry, bar_index, tradeEntry, color=color.new(color.gray, 20), width=3)
        slLine := line.new(bar_index, tradeSL, bar_index, tradeSL, color=color.red, width=3)
        t1Line := line.new(bar_index, tradeT1, bar_index, tradeT1, color=color.new(color.teal, 20), width=3)
        t2Line := line.new(bar_index, tradeT2, bar_index, tradeT2, color=color.new(color.teal, 20), width=3)
        entryLbl := levelLabel(tradeEntry, levelText("Entry", "", tradeEntry), color.gray)
        slLbl := levelLabel(tradeSL, levelText("SL", "", tradeSL), color.red)
        t1Lbl := levelLabel(tradeT1, levelText("T1", "", tradeT1), color.teal)
        t2Lbl := levelLabel(tradeT2, levelText("T2", "", tradeT2), color.teal)
        keepLine(entryLine)
        keepLine(slLine)
        keepLine(t1Line)
        keepLine(t2Line)
        keepLabel(entryLbl, tradeCount)
        keepLabel(slLbl, tradeCount)
        keepLabel(t1Lbl, tradeCount)
        keepLabel(t2Lbl, tradeCount)
    if eraseOld
        pruneDrawings()

// EXITS (simulated, display only)
bool t4Closed = false
bool slClosed = false

if barstate.isconfirmed and tradeDir != 0 and bar_index > tradeEntryBar
    bool stopHit = tradeDir == 1 ? close <= tradeSL : close >= tradeSL
    bool hit1 = tradeDir == 1 ? high >= tradeT1 : low <= tradeT1
    bool hit2 = tradeDir == 1 ? high >= tradeT2 : low <= tradeT2
    bool hit3 = tradeDir == 1 ? high >= tradeT3 : low <= tradeT3
    bool hit4 = tradeDir == 1 ? high >= tradeT4 : low <= tradeT4

    // The stop is close-based: a wick through it does not close the trade, a candle closing beyond it
    // does, at that candle's close. Before the first target is touched, the stop is checked first (the
    // conservative choice): a bar that closes beyond the stop and also touches a target is closed by the
    // stop. Once the first target has been touched, the stop moves to the entry plus the
    // Breakeven Buffer, also close-based, and only the breakeven stop, an ATR flip or the last target
    // closes the trade.
    if stopHit and not t1Hit
        lossCount += 1
        if showTargets
            label.set_text(slLbl, levelText("SL", " ✗", tradeSL) + (maxLosses > 0 and lossCount == maxLosses ? "\nDone for the day" : ""))
        tradeDir := 0
        slClosed := true
    else
        if not t1Hit and hit1
            t1Hit := true
            bookedFrac += 0.25
            bookedR += 0.25 * rResult(tradeT1, tradeDir)
            if showTargets
                label.set_text(t1Lbl, levelText("T1", " ✓", tradeT1))
                // The stop line ends on this candle and the stop continues as a dotted line at the breakeven
                // level, with its label moved there.
                beLine := line.new(bar_index, tradeBE, bar_index, tradeBE, color=color.red, width=3, style=line.style_dotted)
                keepLine(beLine)
                label.set_y(slLbl, tradeBE)
                label.set_text(slLbl, levelText("BE SL", "", tradeBE))
                t3Line := line.new(tradeEntryBar, tradeT3, bar_index, tradeT3, color=color.new(color.teal, 20), width=3)
                t3Lbl := levelLabel(tradeT3, levelText("T3", "", tradeT3), color.teal)
                keepLine(t3Line)
                keepLabel(t3Lbl, tradeCount)

        if not t2Hit and hit2
            t2Hit := true
            bookedFrac += 0.25
            bookedR += 0.25 * rResult(tradeT2, tradeDir)
            if showTargets
                label.set_text(t2Lbl, levelText("T2", " ✓", tradeT2))
                t4Line := line.new(tradeEntryBar, tradeT4, bar_index, tradeT4, color=color.new(color.teal, 20), width=3)
                t4Lbl := levelLabel(tradeT4, levelText("T4", "", tradeT4), color.teal)
                keepLine(t4Line)
                keepLabel(t4Lbl, tradeCount)

        if not t3Hit and hit3
            t3Hit := true
            bookedFrac += 0.25
            bookedR += 0.25 * rResult(tradeT3, tradeDir)
            if showTargets
                label.set_text(t3Lbl, levelText("T3", " ✓", tradeT3))

        if not t4Hit and hit4
            t4Hit := true
            if showTargets
                label.set_text(t4Lbl, levelText("T4", " ✓", tradeT4))
            tradeDir := 0
            t4Closed := true

        if tradeDir != 0 and t1Hit and (tradeDir == 1 ? close <= tradeBE : close >= tradeBE)
            float beR = bookedR + (1 - bookedFrac) * rResult(close, tradeDir)
            if beR < 0
                lossCount += 1
            if showTargets
                label.set_text(slLbl, levelText("BE SL", " ✗", tradeBE) + (maxLosses > 0 and beR < 0 and lossCount == maxLosses ? "\nDone for the day" : ""))
            tradeDir := 0
            slClosed := true

// Closes by stop, last target or cutoff, the candle that touches the first target, and the latest candle.
if showTargets and wasOpen and not flipClosed and tradeEntryBar < bar_index and not na(slLine) and (tradeDir == 0 or barstate.islast or (t1Hit and not t1WasHit))
    extendLevels(not t1WasHit)

// A trade closed on this candle by its stop or last target keeps only the label that shows how it ended.
if showTargets and wasOpen and not flipClosed and tradeDir == 0
    clearLevelLabels(slClosed, t4Closed)

// ADD: the Continuation setup while a trade opened on an earlier candle is still open in the same
// direction after this candle's exits. It opens no new simulated trade and leaves the open trade's
// levels and exits unchanged. It is capped by Max ADDs Per Trade, uses one of the regime's
// continuation re-entries and restarts the cooldown.
addBuy = barstate.isconfirmed and lossLimitOK and useContinuation and smartBuyOn and tradeDir == 1 and tradeEntryBar < bar_index and not na(atr) and prevLongPullback and longResume and continuationCount < maxReentries and addCount < maxAdds and longRSI and cooldownOK
addSell = barstate.isconfirmed and lossLimitOK and useContinuation and smartSellOn and tradeDir == -1 and tradeEntryBar < bar_index and not na(atr) and prevShortPullback and shortResume and continuationCount < maxReentries and addCount < maxAdds and shortRSI and cooldownOK

if addBuy or addSell
    lastSignalBar := bar_index
    continuationCount += 1
    addCount += 1

if addBuy
    longPullbackSeen := false

if addSell
    shortPullbackSeen := false

// ATR FLIP LABELS
if showFlipLabels and flipClosed
    bool bullClose = flipCloseDir == 1
    bool reversed = bullClose ? buySignal : sellSignal
    string outcome = na(flipR) ? "result n/a" : flipR > 0 ? "profit" : flipR < 0 ? "loss" : "flat"
    string flipText = "Flip exit: " + outcome + bookedText(flipBooked) + (reversed ? "\nNow " + (bullClose ? "BUY" : "SELL") : "") + (flipReachedLimit ? "\nDone for the day" : "")
    color flipColor = na(flipR) ? color.gray : flipR > 0 ? color.green : flipR < 0 ? color.red : color.gray
    keepLabel(label.new(bar_index, bullClose ? low : high, flipText, style=bullClose ? label.style_label_up : label.style_label_down, color=color.new(flipColor, 20), textcolor=color.white, size=size.tiny), flipTradeNum)

showAids = input.bool(false, "Show Discretionary Aids", group="Discretionary Aids", tooltip="Shows the trend panel (trend reading, higher-timeframe trend and choppy warning), adds the trend reading to C BUY and C SELL labels (Strong, Neutral or Weak), adds HTF ✓ to C BUY, C SELL and ADD labels when the higher timeframe's ATR trend agrees with the signal, and adds ADX and the Efficiency Ratio to the Data Window. With this off, C BUY and C SELL labels are solid green and red. Signals, levels, exits and alerts are the same either way.")
shadeWeak = input.bool(false, "Shade Weak Candles", group="Discretionary Aids", tooltip="Light grey background on closed candles rated Weak. Needs Show Discretionary Aids on.")

// TREND STRENGTH
dmiLen = input.int(14, "DMI Length", minval=1, group="Trend Strength", tooltip="Length for +DI, -DI and the ADX smoothing.")
adxWeak = input.float(18, "Weak Below ADX", step=0.5, minval=0, group="Trend Strength", tooltip="Weak when ADX is below this level, or when the Efficiency Ratio is below the Weak Below Efficiency level.")
adxStrong = input.float(22, "Strong Above ADX", step=0.5, minval=0, group="Trend Strength", tooltip="Strong when ADX is above this level and higher than 3 bars earlier, and the leading DI matches the ATR trail's direction. Anything between Weak and Strong is Building.")
erLen = input.int(20, "Efficiency Window (candles)", minval=2, maxval=100, group="Trend Strength", tooltip="Used on timeframes other than 5m and 15m. On 5m charts the window is 24 candles and on 15m charts 12.")
erWeak = input.float(0.3, "Weak Below Efficiency", step=0.05, minval=0, maxval=1, group="Trend Strength", tooltip="Weak, and the choppy warning shows, when the Efficiency Ratio is below this level. The Efficiency Ratio is the net move over the window divided by the total distance price travelled in it: near 1 when price moves in one direction, near 0 when it swings back and forth.")

[diPlus, diMinus, adx] = ta.dmi(dmiLen, dmiLen)
adxRising = adx > adx[3]
diLead = diPlus > diMinus ? 1 : diMinus > diPlus ? -1 : 0

// Efficiency window: 24 candles on 5m charts, 12 on 15m charts, the input on other timeframes.
int effErLen = timeframe.isminutes and timeframe.multiplier == 5 ? 24 : timeframe.isminutes and timeframe.multiplier == 15 ? 12 : erLen

// EFFICIENCY RATIO: the net move over the window divided by the total distance travelled in it,
// over a rolling window.
float erPath = math.sum(math.abs(close - close[1]), effErLen)
float er = na(erPath) ? na : erPath > 0 ? math.abs(close - close[effErLen]) / erPath : 0.0

// -1 = not enough data yet, 0 = Weak, 1 = Building, 2 = Strong
int strength = na(adx) or na(er) ? -1 : adx < adxWeak or er < erWeak ? 0 : adx > adxStrong and adxRising and atrRegime != 0 and diLead == atrRegime ? 2 : 1

// HIGHER TIMEFRAME TREND
// The ATR trend of the next higher timeframe with this chart's ATR settings, read from its last candle that
// closed before the current candle opened, so it never repaints. 1m charts read 5m, up to 5m read 15m, up to
// 15m read 1h, up to 1h read 4h, other intraday charts read daily, daily read weekly, weekly read monthly
// and monthly read 3 months.
string htfTf = timeframe.isintraday ? (timeframe.in_seconds() <= 60 ? "5" : timeframe.in_seconds() <= 300 ? "15" : timeframe.in_seconds() <= 900 ? "60" : timeframe.in_seconds() <= 3600 ? "240" : "D") : timeframe.isdaily ? "W" : timeframe.isweekly ? "M" : "3M"
string htfName = htfTf == "5" ? "5m" : htfTf == "15" ? "15m" : htfTf == "60" ? "1h" : htfTf == "240" ? "4h" : htfTf

// ATR trend direction, built the same way as the chart's ATR trail: 1 up, -1 down, 0 before the first flip.
atrDirection() =>
    float stopDist = atrMultiplier * ta.atr(atrPeriod)
    var float trail = na
    if na(trail[1])
        trail := close - stopDist
    else if close > trail[1] and close[1] > trail[1]
        trail := math.max(trail[1], close - stopDist)
    else if close < trail[1] and close[1] < trail[1]
        trail := math.min(trail[1], close + stopDist)
    else if close > trail[1]
        trail := close - stopDist
    else
        trail := close + stopDist
    var int dir = 0
    if close[1] < trail[1] and close > trail[1]
        dir := 1
    else if close[1] > trail[1] and close < trail[1]
        dir := -1
    dir

int htfDir = request.security(syminfo.tickerid, htfTf, atrDirection()[1], lookahead=barmerge.lookahead_on)

if showAids and barstate.islast
    bool live = not barstate.isconfirmed
    int shownStrength = live ? strength[1] : strength
    bool shownChoppy = live ? choppyNow[1] : choppyNow
    int shownHtf = live ? htfDir[1] : htfDir
    string stateText = shownStrength == 2 ? "Strong" : shownStrength == 1 ? "Building" : shownStrength == 0 ? "Weak" : "n/a"
    color stateColor = shownStrength == 2 ? color.new(color.green, 20) : shownStrength == 1 ? color.new(color.orange, 20) : color.new(color.gray, 20)
    string htfText = shownHtf == 1 ? "Up" : shownHtf == -1 ? "Down" : "n/a"
    color htfColor = shownHtf == 1 ? color.new(color.green, 20) : shownHtf == -1 ? color.new(color.red, 20) : color.new(color.gray, 20)
    table.cell(strengthPanel, 0, 0, "Trend", text_color=color.white, text_size=size.small)
    table.cell(strengthPanel, 1, 0, stateText, bgcolor=stateColor, text_color=color.white, text_size=size.small)
    table.cell(strengthPanel, 0, 1, "HTF " + htfName, text_color=color.white, text_size=size.small)
    table.cell(strengthPanel, 1, 1, htfText, bgcolor=htfColor, text_color=color.white, text_size=size.small)
    if shownChoppy
        table.cell(strengthPanel, 0, 2, "Market", text_color=color.white, text_size=size.small)
        table.cell(strengthPanel, 1, 2, "⚠ Choppy", bgcolor=color.new(color.yellow, 0), text_color=color.black, text_size=size.small)
    else
        table.clear(strengthPanel, 0, 2, 1, 2)
