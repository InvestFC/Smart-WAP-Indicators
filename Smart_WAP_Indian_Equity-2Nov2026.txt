indicator("Smart WAP Indian Equity", "WAP Equity", overlay=true, max_labels_count=500, max_lines_count=500)

// ATR
atrPeriod = input.int(10, "ATR Period", minval=1, group="ATR Trailing Stop", tooltip="On 1m to 4m charts, the session's first candle counts only its own high-low range, so an overnight gap does not widen the trail, stop or targets.")
atrMultiplier = input.float(4.0, "ATR Multiplier", step=0.1, minval=0, group="ATR Trailing Stop", tooltip="Trailing-stop distance in ATRs. Higher = fewer, slower trend flips.")

// On 1m to 4m charts the session's first candle counts only its own high-low range, so the overnight gap does not enter ATR.
// The higher-timeframe read keeps the standard ATR, so it matches that timeframe's own chart.
bool subFive = timeframe.isminutes and timeframe.multiplier < 5
float rawTrueRange = ta.tr(true)
float atrTrueRange = subFive and session.isfirstbar_regular ? high - low : rawTrueRange
float atrGapFree = ta.rma(atrTrueRange, atrPeriod)
float atrStandard = ta.atr(atrPeriod)
atr = subFive ? atrGapFree : atrStandard
atrStop = atrMultiplier * atr

var float atrTrail = na
if na(atrTrail[1])
    atrTrail := close - atrStop
else if close > atrTrail[1] and close[1] > atrTrail[1]
    atrTrail := math.max(atrTrail[1], close - atrStop)
else if close < atrTrail[1] and close[1] < atrTrail[1]
    atrTrail := math.min(atrTrail[1], close + atrStop)
else if close > atrTrail[1]
    atrTrail := close - atrStop
else
    atrTrail := close + atrStop

var int atrRegime = 0
if na(atrRegime[1])
    atrRegime := 0
else if close[1] < atrTrail[1] and close > atrTrail[1]
    atrRegime := 1
else if close[1] > atrTrail[1] and close < atrTrail[1]
    atrRegime := -1
else
    atrRegime := atrRegime[1]

bullFlip = barstate.isconfirmed and atrRegime == 1 and atrRegime[1] == -1
bearFlip = barstate.isconfirmed and atrRegime == -1 and atrRegime[1] == 1

var int tradeDir = 0
var float tradeEntry = na
var float tradeSL = na
var float tradeBE = na
var int tradeEntryBar = na
var bool t1Hit = false
var float bookedFrac = 0.0
var float bookedR = 0.0
var int tradeCount = 0

// Booking model: touching a target books 25% of the position there, and the last target closes
// whatever remains. Once the first target is touched, the stop moves to the entry plus the
// Breakeven Buffer, and the rest closes at that stop, at an ATR flip or at the last target. The
// result of a closed trade in R blends the booked parts with the exit of the remainder.

// Result of a closed trade in R, where 1R is the initial stop distance (entry to stop).
rResult(float exitPrice, int dir) =>
    float risk = math.abs(tradeEntry - tradeSL)
    risk > 0 ? dir * (exitPrice - tradeEntry) / risk : na

bookedText(float frac) =>
    frac > 0 ? "\n" + str.tostring(math.round(frac * 100)) + "% booked" : ""

// Captured before any closure path (ATR flip, SL, T4, cutoff) or target check runs this bar.
wasOpen = tradeDir != 0
t1WasHit = t1Hit

bool flipClosed = false
int flipCloseDir = 0
float flipR = na
float flipBooked = na
int flipTradeNum = na

// Before the first target is touched, a candle that closes beyond the stop closes the trade at that close,
// even if the same candle also closes across the ATR trail, so the flip does not close it here.
bool stopCrossed = tradeDir != 0 and not t1Hit and bar_index > tradeEntryBar and (tradeDir == 1 ? close <= tradeSL : close >= tradeSL)

if bullFlip and tradeDir == -1 and not stopCrossed
    flipR := bookedR + (1 - bookedFrac) * rResult(close, tradeDir)
    flipBooked := bookedFrac
    flipTradeNum := tradeCount
    tradeDir := 0
    flipClosed := true
    flipCloseDir := 1

if bearFlip and tradeDir == 1 and not stopCrossed
    flipR := bookedR + (1 - bookedFrac) * rResult(close, tradeDir)
    flipBooked := bookedFrac
    flipTradeNum := tradeCount
    tradeDir := 0
    flipClosed := true
    flipCloseDir := -1

// Each regime has its own line, and a line breaks where its regime ends. A flip therefore shows
// a clean step to the other side in the other colour, with no diagonal connecting the two.
plot(atrRegime == 1 ? atrTrail : na, "ATR Trailing Stop (up)", color=color.green, linewidth=1, style=plot.style_linebr)
plot(atrRegime == -1 ? atrTrail : na, "ATR Trailing Stop (down)", color=color.red, linewidth=1, style=plot.style_linebr)
plot(atrRegime == 0 ? atrTrail : na, "ATR Trailing Stop (before first flip)", color=color.navy, linewidth=1, style=plot.style_linebr)

Changing it away from EMA changes the signal logic, not just the chart overlay. VWMA needs volume data and returns nothing on symbols without it.")

fastColor = input.color(color.yellow, "Fast MA", inline="maColours1", group="Moving Averages")
slowColor = input.color(color.purple, "Slow MA", inline="maColours1", group="Moving Averages")

ma(series float source, int length) =>
    float result = na
    if maType == "EMA"
        result := ta.ema(source, length)
    else if maType == "SMA"
        result := ta.sma(source, length)
    else if maType == "WMA"
        result := ta.wma(source, length)
    else if maType == "RMA"
        result := ta.rma(source, length)
    else if maType == "VWMA"
        result := ta.vwma(source, length)
    else
        result := ta.hma(source, length)
    result

fastMA = ma(close, fastLen)
slowMA = ma(close, slowLen)
ma20 = showExtraMAs ? ma(close, 20) : na
ma50 = showExtraMAs ? ma(close, 50) : na
ma100 = showExtraMAs ? ma(close, 100) : na
ma200 = showExtraMAs ? ma(close, 200) : na

plot(showFastSlow ? fastMA : na, "Fast MA", color=fastColor, linewidth=2)
plot(showFastSlow ? slowMA : na, "Slow MA", color=slowColor, linewidth=2)
plot(ma20, "20 MA", color=ma20Color)
plot(ma50, "50 MA", color=ma50Color)
plot(ma100, "100 MA", color=ma100Color)
plot(ma200, "200 MA", color=ma200Color, linewidth=2)

smartSellOn = input.bool(true, "Smart Sell Signal", group="Signals")

minSignalGap = input.int(10, "Minimum Bars Between Signals", minval=0, group="Signal Filter", tooltip="Minimum bars between any two signals (trend or continuation).")

rsi = ta.rsi(close, rsiLen)

// TARGETS (ATR-based, informational only, not orders)
slMult = input.float(2.7, "Stop Loss ATR Multiplier", step=0.1, minval=0, group="SL / Targets")

t1Mult = input.float(1.5, "Target 1 ATR Multiplier", step=0.1, minval=0, group="SL / Targets")

showTargets = input.bool(true, "Show Target/SL Labels and Lines", group="SL / Targets", tooltip="Levels are informational only. The entry line marks the confirmed signal bar's close, and the stop and targets are computed from that close and ATR and frozen for that simulated trade. Touching a target books 25% of the position there and ticks its label, and the last target closes the rest. Before the first target, the trade closes when a candle closes beyond the stop; a wick through it does not count. After the first target, the stop moves to the entry plus the Breakeven Buffer, and an ATR flip also closes what remains. Raise it if you raise Max Continuation Re-Entries.")

// Draws a trade label. 'above' puts the label above the level, pointing down.
tradeLabel(float y, string txt, bool above, color c) =>
    if showTargets
        label.new(bar_index, y, txt, style=above ? label.style_label_down : label.style_label_up, color=color.new(c, 20), textcolor=color.white, size=size.tiny)

// Text for a level's price label: the name and price while the level is live, or the name and a mark once a
// target is touched (tick) or a stop closes the trade (cross).
levelText(string name, string mark, float price) =>
    mark == "" ? name + ": " + str.tostring(price, format.mintick) : name + mark

// Price label at the right end of a level line. It moves along with the line as the line extends.
levelLabel(float y, string txt, color c) =>
    label.new(bar_index, y, txt, style=label.style_label_left, color=color.new(c, 20), textcolor=color.white, size=size.tiny)

// SESSION CUTOFF
useCutoff = input.bool(true, "Stop Signals at Session Cutoff", group="Session Cutoff", tooltip="Blocks new signals from the cutoff time onward and closes the simulated trade (informational only, no real position exists). Uses the exchange timezone, so 15:15 is IST. 15:15 lines up with a bar close on every common NSE timeframe (1m to 60m). Also resets pending MA confirmation and continuation pullbacks at each new session.")
cutoffHour = input.int(15, "Cutoff Hour", minval=0, maxval=23, group="Session Cutoff")
cutoffMinute = input.int(15, "Cutoff Minute", minval=0, maxval=59, group="Session Cutoff", tooltip="Compared with each bar's close time. On timeframes where no bar closes exactly at the cutoff, it fires on the next bar close after it.")

// Stateless: recomputed from the bar's own date every bar, so it can never get stuck on.
sessionCutoff = timestamp(syminfo.timezone, year(time, syminfo.timezone), month(time, syminfo.timezone), dayofmonth(time, syminfo.timezone), cutoffHour, cutoffMinute)
atCutoff = useCutoff and time_close >= sessionCutoff

// GAP FADE
useGapFade = input.bool(true, "Enable Gap Fade Signal", group="Gap Fade Signal", tooltip="Looks for a gap at the session open that fails to hold and reverses back through the opening price within the window below. Shares the cooldown with the other signals and ignores the RSI filter. On continuous futures symbols, the jump at contract expiry can be read as a gap.")
gapPctMin = input.float(0.3, "Minimum Gap Size (%)", step=0.05, minval=0, group="Gap Fade Signal", tooltip="The session open must differ from the previous regular-session close by at least this percentage. Individual stocks usually gap more than an index, so you may want a higher value for them.")
gapWindow = input.int(3, "Fade Confirmation Window (bars)", minval=1, group="Gap Fade Signal", tooltip="How many bars after the opening bar to watch for the reversal back through the open. The opening bar itself cannot show a reversal, so the window starts on the next bar. On 1m to 4m charts the window covers the same time as this many 15m candles: 3 covers 45 minutes, which is 15 candles on 3m.")

newSession = session.isfirstbar_regular
// Overnight state resets at a session boundary when sessions matter (cutoff or Gap Fade on).
resetOnSession = newSession and (useCutoff or useGapFade)

var float lastRTHClose = na
if session.islastbar_regular
    lastRTHClose := close

var float sessionOpen = na
var int sessionOpenBar = na
var int gapDirection = 0
var bool gapDone = false

if newSession
    float sessionClose = na(lastRTHClose) ? close[1] : lastRTHClose
    sessionOpen := open
    sessionOpenBar := bar_index
    gapDone := false

    float gap = 0.0
    if not na(sessionClose) and sessionClose != 0
        gap := (sessionOpen - sessionClose) / sessionClose * 100

    if gap >= gapPctMin
        gapDirection := 1
    else if gap <= -gapPctMin
        gapDirection := -1
    else
        gapDirection := 0

// The opening bar itself cannot show a reversal through the open, so the window starts on the next bar.
barsFromOpen = na(sessionOpenBar) ? na : bar_index - sessionOpenBar
// On 1m to 4m charts the window covers the same time as Fade Confirmation Window 15m candles.
int effGapWindow = subFive ? math.round(gapWindow * 900.0 / timeframe.in_seconds()) : gapWindow
inGapWindow = not na(barsFromOpen) and barsFromOpen >= 1 and barsFromOpen <= effGapWindow

gapFadeBuy = useGapFade and gapDirection == -1 and inGapWindow and not gapDone and not atCutoff and close > sessionOpen
gapFadeSell = useGapFade and gapDirection == 1 and inGapWindow and not gapDone and not atCutoff and close < sessionOpen

// STATE
var float tradeT1 = na
var float tradeT2 = na
var float tradeT3 = na
var float tradeT4 = na

var bool t2Hit = false
var bool t3Hit = false
var bool t4Hit = false

var line entryLine = na
var line slLine = na
var line beLine = na
var line t1Line = na
var line t2Line = na
var line t3Line = na
var line t4Line = na

var label entryLbl = na
var label slLbl = na
var label t1Lbl = na
var label t2Lbl = na
var label t3Lbl = na
var label t4Lbl = na

var int pendingDir = 0
var int pendingBar = na

var bool longPullbackSeen = false
var bool shortPullbackSeen = false
var int lastFlipDir = 0
var int continuationCount = 0
// ADD signals given to the open trade; reset at every entry.
var int addCount = 0
// Direction of the latest trend signal. Cleared at every ATR flip and at the session reset; 0 = none since then.
var int freshTrendDir = 0
// True once an ATR flip has occurred since the latest session reset. While false, the trend was carried over from an earlier session.
var bool flipSinceReset = false

var int lastSignalBar = na

// CONTINUATION INPUTS
useContinuation = input.bool(true, "Enable Continuation Signal", group="Continuation Signal", tooltip="Re-enters a still-running trend after a pullback to the Fast MA followed by a resume candle, without needing a fresh ATR flip. It also needs a trend BUY (for C BUY) or SELL (for C SELL) earlier in the same session and after the latest ATR flip, so a trend carried over from an earlier session gives no continuation entries until a fresh trend signal fires, unless Allow Continuation in Carried Trend is on. The first candle of each session cannot count as the pullback. Continuation entries are labelled C BUY and C SELL. When a simulated trade is already open in the same direction, the same setup shows ADD instead: it opens no new trade, leaves the open trade's levels and exits unchanged, and uses one of the re-entries. With Show Discretionary Aids on, the label adds the trend panel's reading on the signal candle: Strong (solid), Neutral for Building or Opening (lighter) or Weak (faded). With it off the label is plain and solid. C BUY, C SELL and ADD labels also show HTF ✓ when the higher timeframe's ATR trend agrees with the signal.")
allowCarry = input.bool(true, "Allow Continuation in Carried Trend", group="Continuation Signal", tooltip="Lets the Continuation setup fire in a trend carried over from an earlier session, before any ATR flip in the current session. These entries are labelled Carry BUY and Carry SELL and have their own alerts. Each opens a normal simulated trade with the usual stop, targets and exits, and uses the cooldown, the continuation re-entries and the losing-trade limit like any Continuation entry. While that trade is open, the same setup shows ADD. After an ATR flip in the session, the usual rule applies again. Set it before the session, not during it.")
maxReentries = input.int(3, "Max Continuation Re-Entries Per Regime", minval=1, group="Continuation Signal", tooltip="Cap on continuation entries, ADD signals included, within one ATR regime. Resets on every ATR flip and at every session reset.")
maxAdds = input.int(1, "Max ADDs Per Trade", minval=0, group="Continuation Signal", tooltip="How many ADD signals one open simulated trade can get. Set this to your own risk appetite: every add you take raises your real risk on the trade above what its stop and targets show. 0 turns ADD off. ADD signals also count toward Max Continuation Re-Entries Per Regime.")

// SESSION RESET
// The re-entry count restarts with the session, so a carried trend starts the day with its full allowance.
if resetOnSession
    freshTrendDir := 0
    flipSinceReset := false
    continuationCount := 0
    pendingDir := 0
    pendingBar := na
    longPullbackSeen := false
    shortPullbackSeen := false

// FLIPS
if bullFlip
    lastFlipDir := 1
    flipSinceReset := true
    continuationCount := 0
    freshTrendDir := 0
    longPullbackSeen := false
    shortPullbackSeen := false
    if useMAConfirm
        pendingDir := 1
        pendingBar := bar_index
else if bearFlip
    lastFlipDir := -1
    flipSinceReset := true
    continuationCount := 0
    freshTrendDir := 0
    longPullbackSeen := false
    shortPullbackSeen := false
    if useMAConfirm
        pendingDir := -1
        pendingBar := bar_index

// MA PENDING WINDOW
pendingAge = pendingDir != 0 and not na(pendingBar) ? bar_index - pendingBar : na

if pendingDir != 0 and pendingAge > maxPendingBars
    pendingDir := 0
    pendingBar := na

bool longMAReady = bullFlip
bool shortMAReady = bearFlip

if useMAConfirm
    longMAReady := pendingDir == 1 and pendingAge >= 1 and fastMA > slowMA
    shortMAReady := pendingDir == -1 and pendingAge >= 1 and fastMA < slowMA

longRSI = not useRSI or rsi >= buyRSI
shortRSI = not useRSI or rsi <= sellRSI

// CONTINUATION: pullback to the Fast MA, then a resume candle
// The first candle of a session never counts as a pullback, because the overnight gap distorts it.
longPullback = barstate.isconfirmed and not newSession and atrRegime == 1 and lastFlipDir == 1 and pendingDir != 1 and low <= fastMA and close < fastMA
shortPullback = barstate.isconfirmed and not newSession and atrRegime == -1 and lastFlipDir == -1 and pendingDir != -1 and high >= fastMA and close > fastMA

longResume = barstate.isconfirmed and atrRegime == 1 and lastFlipDir == 1 and pendingDir != 1 and close > fastMA and close > open
shortResume = barstate.isconfirmed and atrRegime == -1 and lastFlipDir == -1 and pendingDir != -1 and close < fastMA and close < open

if longPullback
    longPullbackSeen := true

if shortPullback
    shortPullbackSeen := true

// [1] reads the previous bar's value, so a flip or session reset must also block it explicitly.
prevLongPullback = longPullbackSeen[1] and not bullFlip and not resetOnSession
prevShortPullback = shortPullbackSeen[1] and not bearFlip and not resetOnSession

// CARRIED TREND: no ATR flip since the latest session reset, so the regime was carried over from an earlier session.
carryLong = allowCarry and not flipSinceReset and lastFlipDir == 1
carryShort = allowCarry and not flipSinceReset and lastFlipDir == -1

// LOSING TRADES PER SESSION
// Simulated trades closed at a net loss in the current session. An ATR flip exit is counted here, before the
// signals, so it also blocks a re-entry on the same candle. Stop and cutoff exits are counted where they happen.
var int lossCount = 0
if newSession
    lossCount := 0
if flipClosed and flipR < 0
    lossCount += 1
// True when this candle's ATR flip exit is the losing trade that reaches the limit.
bool flipReachedLimit = maxLosses > 0 and flipClosed and flipR < 0 and lossCount == maxLosses
bool lossLimitOK = maxLosses == 0 or lossCount < maxLosses

// SIGNALS
tradeFlat = tradeDir == 0
cooldownOK = na(lastSignalBar) or bar_index - lastSignalBar >= minSignalGap

trendBuy = barstate.isconfirmed and lossLimitOK and smartBuyOn and longMAReady and longRSI and tradeFlat and cooldownOK and not atCutoff and not na(atr)
trendSell = barstate.isconfirmed and lossLimitOK and smartSellOn and shortMAReady and shortRSI and tradeFlat and cooldownOK and not atCutoff and not na(atr)

contBuy = barstate.isconfirmed and lossLimitOK and useContinuation and smartBuyOn and tradeFlat and not na(atr) and (freshTrendDir == 1 or carryLong) and prevLongPullback and longResume and continuationCount < maxReentries and longRSI and cooldownOK and not atCutoff
contSell = barstate.isconfirmed and lossLimitOK and useContinuation and smartSellOn and tradeFlat and not na(atr) and (freshTrendDir == -1 or carryShort) and prevShortPullback and shortResume and continuationCount < maxReentries and shortRSI and cooldownOK and not atCutoff

gapBuy = barstate.isconfirmed and lossLimitOK and smartBuyOn and gapFadeBuy and tradeFlat and cooldownOK and not na(atr)
gapSell = barstate.isconfirmed and lossLimitOK and smartSellOn and gapFadeSell and tradeFlat and cooldownOK and not na(atr)

// Priority when candidates coincide: trend (3), continuation (2), gap fade (1).
// The higher priority wins; equal priority in opposite directions cancels out.
int buyPriority = 0
int sellPriority = 0

if trendBuy
    buyPriority := 3
else if contBuy
    buyPriority := 2
else if gapBuy
    buyPriority := 1

if trendSell
    sellPriority := 3
else if contSell
    sellPriority := 2
else if gapSell
    sellPriority := 1

buySignal = buyPriority > 0 and buyPriority > sellPriority
sellSignal = sellPriority > 0 and sellPriority > buyPriority

trendBuySig = buySignal and buyPriority == 3
trendSellSig = sellSignal and sellPriority == 3
contBuySig = buySignal and buyPriority == 2
contSellSig = sellSignal and sellPriority == 2
// Continuation entries taken in a carried trend, labelled Carry BUY and Carry SELL.
carryBuySig = contBuySig and freshTrendDir != 1
carrySellSig = contSellSig and freshTrendDir != -1
gapBuySig = buySignal and buyPriority == 1
gapSellSig = sellSignal and sellPriority == 1

if buySignal or sellSignal
    lastSignalBar := bar_index

if trendBuySig or trendSellSig
    freshTrendDir := trendBuySig ? 1 : -1
    pendingDir := 0
    pendingBar := na

if contBuySig
    continuationCount += 1
    longPullbackSeen := false

if contSellSig
    continuationCount += 1
    shortPullbackSeen := false

// The Gap Fade opportunity is used up only when a Gap Fade signal actually fires.
if gapBuySig or gapSellSig
    gapDone := true

// TRADE DRAWING HISTORY
// Each trade's drawings are stored with the trade's number so the oldest trades can be deleted.
var array<line> drawnLines = array.new<line>()
var array<int> lineOwner = array.new<int>()
var array<label> drawnLabels = array.new<label>()
var array<int> labelOwner = array.new<int>()

keepLine(line l) =>
    if eraseOld
        array.push(drawnLines, l)
        array.push(lineOwner, tradeCount)

keepLabel(label l, int owner) =>
    if eraseOld
        array.push(drawnLabels, l)
        array.push(labelOwner, owner)

// Deletes the drawings of every trade older than the most recent Trades to Keep Drawn.
pruneDrawings() =>
    int oldest = tradeCount - keepTrades + 1
    if array.size(drawnLines) > 0

                array.remove(labelOwner, i)

// On intraday charts of 15 minutes or less the window stops at the session's first candle, and that candle's
// move is measured from its own open, so the overnight gap is never counted. The reading is ready once the

// ATR trend direction, built the same way as the chart's ATR trail: 1 up, -1 down, 0 before the first flip.
atrDirection() =>
    float stopDist = atrMultiplier * ta.atr(atrPeriod)
    var float trail = na
    if na(trail[1])
        trail := close - stopDist
    else if close > trail[1] and close[1] > trail[1]
        trail := math.max(trail[1], close - stopDist)
    else if close < trail[1] and close[1] < trail[1]
        trail := math.min(trail[1], close + stopDist)
    else if close > trail[1]
        trail := close - stopDist
    else
        trail := close + stopDist
    var int dir = 0
    if close[1] < trail[1] and close > trail[1]
        dir := 1
    else if close[1] > trail[1] and close < trail[1]
        dir := -1
    dir
